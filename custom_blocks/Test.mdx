import { useState } from 'react';

export const SdkSnippet = ({ lang = 'python', install, children }) => {
  const [copied, setCopied] = useState(false);

  const code = (typeof children === 'string'
    ? children
    : children?.props?.children ?? ''
  ).toString().trim();

  const copy = () => {
    navigator.clipboard.writeText(code).then(() => {
      setCopied(true);
      setTimeout(() => setCopied(false), 1500);
    });
  };

  return (
    <div className="rounded-lg overflow-hidden border border-gray-700 bg-[#1e1f23] my-4">
      <div className="flex items-center justify-between px-4 py-2 bg-[#2a2b31] border-b border-gray-700">
        <span className="text-xs uppercase tracking-wide text-gray-400">{lang}</span>
        <button onClick={copy} className="text-xs text-gray-400 hover:text-white">
          {copied ? 'Copied' : 'Copy'}
        </button>
      </div>

      {install && (
        <div className="px-4 py-2 bg-[#26272c] border-b border-gray-700 font-mono text-sm text-green-400">
          <span className="text-gray-500 mr-2">$</span>{install}
        </div>
      )}

      <pre className="px-4 py-3 overflow-x-auto text-sm leading-relaxed text-gray-200">
        <code>{code}</code>
      </pre>
    </div>
  );
};

<SdkSnippet lang="javascript">
  {`const hello = "world";

console.log(hello);`}
</SdkSnippet>
